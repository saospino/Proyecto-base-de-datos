       1 + (g % 10), 1 + (g % 4),

       DATE '2016-07-01' + (g % 3650) + 15,
       DATE '2016-07-01' + (g % 3650) + 10 + (g % 8)

JOIN LATERAL (SELECT id_libro, num_ejemplar FROM ejemplares
              ORDER BY id_libro, num_ejemplar
